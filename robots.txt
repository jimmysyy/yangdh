# JMNAV 几米导航 — robots.txt
#
# 静态站实际只对外暴露 5 个路径：
#   /               主页
#   /style.css      样式（?v= 内容哈希）
#   /main.js        脚本（?v= 内容哈希）
#   /robots.txt
#   /sitemap.xml
# 其余规则是防御性的：万一误把仓库根目录部署上线，也能拦住源码与文档被索引。

User-agent: *
Allow: /
Disallow: /assets/
Disallow: /tools/
Disallow: /docs/
Disallow: /dist/
Disallow: /*.md
Disallow: /*.bak
Disallow: /package.json
Disallow: /_headers
Disallow: /perf-test.html

Disallow: /

User-agent: DotBot
Disallow: /

User-agent: MJ12bot
Disallow: /

User-agent: YisouSpider
Disallow: /

# ── AI 训练爬虫 ──
User-agent: GPTBot
Disallow: /

User-agent: CCBot
Disallow: /

User-agent: ClaudeBot
Disallow: /

User-agent: anthropic-ai

User-agent: PerplexityBot

# 说明：
# 已删除 Googlebot / Bingbot / Baiduspider 的独立分组——它们原本只写 Allow: /，
# 而独立分组会让 User-agent: * 里的 Disallow 对它们失效（robots.txt 里
# 爬虫只遵守自己那一组）。删掉后上面的防御性 Disallow 才真正对 Google 生效。
# 已删除 Crawl-delay：Google 与 Bing 都已忽略该指令。
